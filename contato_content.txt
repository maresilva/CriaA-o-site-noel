                <span class="ca-btn-text">Enviar Solicitação</span>

              <p style="text-align:center; color: rgba(255,255,255,0.3); font-size:12px; margin-top:16px;">Sua solicitação será enviada por e-mail e WhatsApp para nosso time comercial.</p>

            <!-- Toast Notification -->
            <div class="ca-toast" id="ca-toast"></div>

            <!-- Script: checkbox toggle + dual send (email + WhatsApp) -->
            <script>
              (function() {
                // =============================================
                // Web3Forms config — crie sua chave gratuita em
                // https://web3forms.com/ e substitua abaixo
                // =============================================
                var WEB3FORMS_KEY = 'YOUR_WEB3FORMS_ACCESS_KEY'; // <-- TROQUE AQUI

                // --- Checkbox toggle behavior ---
                var grid = document.getElementById('ca-servicos-grid');
                if (grid) {
                  grid.addEventListener('click', function(e) {
                    var item = e.target.closest('.ca-checkbox-item');
                    if (!item) return;
                    var cb = item.querySelector('input[type="checkbox"]');
                    cb.checked = !cb.checked;
                    item.classList.toggle('checked', cb.checked);
                  });
                }

                // --- Toast helper ---
                function showToast(msg, type) {
                  var toast = document.getElementById('ca-toast');
                  if (!toast) return;
                  toast.textContent = msg;
                  toast.className = 'ca-toast';
                  if (type === 'error') toast.classList.add('ca-toast--error');
                  if (type === 'success') toast.classList.add('ca-toast--success');
                  toast.classList.add('show');
                  clearTimeout(toast._t);
                  toast._t = setTimeout(function() {
                    toast.classList.remove('show');
                  }, 4000);
                }

                // --- Disable / enable button ---
                function setButtonState(btn, state) {
                  if (!btn) return;
                  btn.className = 'ca-form-submit';
                  if (state === 'loading') {
                    btn.classList.add('ca-btn-loading');
                    btn.disabled = true;
                  } else {
                    btn.disabled = false;
                  }
                }

                // --- Get selected services ---
                function getServicos() {
                  var items = document.querySelectorAll('#ca-servicos-grid .ca-checkbox-item.checked');
                  var list = [];
                  items.forEach(function(item) { list.push(item.getAttribute('data-value')); });
                  return list;
                }

                // --- Build plain-text email body ---
                function buildEmailBody(data) {
                  var body = 'NOVA SOLICITAÇÃO DE PROPOSTA — CRIA AÇÃO ENTRETENIMENTO\n';
                  body += '========================================================\n\n';

                  body += '--- DADOS DO SOLICITANTE ---\n';
                  body += 'Nome Comercial: ' + data.nomeComercial + '\n';
                  body += 'Responsável:    ' + data.responsavel + '\n';
                  if (data.cargo)    body += 'Cargo:          ' + data.cargo + '\n';
                  body += 'Telefone:       ' + data.telefone + '\n';
                  if (data.email)    body += 'E-mail:         ' + data.email + '\n';

                  body += '\n--- DADOS DO EVENTO ---\n';
                  body += 'Data Início:    ' + data.dataInicio + '\n';
                  body += 'Data Término:   ' + data.dataTermino + '\n';
                  if (data.horaInicio)  body += 'Início:         ' + data.horaInicio + '\n';
                  if (data.horaTermino) body += 'Término:        ' + data.horaTermino + '\n';
                  body += 'Cidade/Estado:  ' + data.cidadeEstado + '\n';

                  if (data.servicos.length > 0) {
                    body += '\n--- SERVIÇOS SOLICITADOS ---\n';
                    data.servicos.forEach(function(s) { body += '• ' + s + '\n'; });
                  }

                  if (data.mensagem) {
                    body += '\n--- OBSERVAÇÕES ---\n';
                    body += data.mensagem + '\n';
                  }

                  body += '\n========================================================\n';
                  body += 'Enviado pelo formulário de contato do site criaacao.com\n';
                  return body;
                }

                // --- Send via Web3Forms API ---
                function sendEmail(data) {
                  if (!WEB3FORMS_KEY || WEB3FORMS_KEY === 'YOUR_WEB3FORMS_ACCESS_KEY') {
                    // Key not configured — skip silently, WhatsApp still works
                    return Promise.resolve({ ok: false, skipped: true });
                  }

                  var formData = new FormData();
                  formData.append('access_key', WEB3FORMS_KEY);
                  formData.append('subject', 'Novo Contato — ' + data.nomeComercial + ' (' + data.cidadeEstado + ')');
                  formData.append('from_name', data.responsavel);
                  formData.append('from_email', data.email || 'formulario@criaacao.com');
                  formData.append('reply_to', data.email || '');
                  formData.append('message', buildEmailBody(data));
                  formData.append('Nome_Comercial', data.nomeComercial);
                  formData.append('Responsavel', data.responsavel);
                  formData.append('Cargo', data.cargo);
                  formData.append('Telefone', data.telefone);
                  formData.append('Data_Inicio', data.dataInicio);
                  formData.append('Data_Termino', data.dataTermino);
                  formData.append('Hora_Inicio', data.horaInicio);
                  formData.append('Hora_Termino', data.horaTermino);
                  formData.append('Cidade_Estado', data.cidadeEstado);
                  formData.append('Servicos', data.servicos.join(', '));
                  formData.append('Observacoes', data.mensagem);

                  return fetch('https://api.web3forms.com/submit', {
                    method: 'POST',
                    body: formData
                  }).then(function(res) { return res.json(); });
                }

                // --- Build WhatsApp URL ---
                function buildWhatsAppUrl(data) {
                  var texto = '*📋 NOVA SOLICITAÇÃO DE PROPOSTA — CRIA AÇÃO ENTRETENIMENTO*\n\n';
                  texto += '*━━━ DADOS DO SOLICITANTE ━━━*\n';
                  texto += '🏢 *Nome Comercial:* ' + data.nomeComercial + '\n';
                  texto += '👤 *Responsável:* ' + data.responsavel + '\n';
                  if (data.cargo) texto += '💼 *Cargo:* ' + data.cargo + '\n';
                  texto += '📱 *Telefone:* ' + data.telefone + '\n';
                  if (data.email) texto += '📧 *E-mail:* ' + data.email + '\n';

                  texto += '\n*━━━ DADOS DO EVENTO ━━━*\n';
                  texto += '📅 *Data Início:* ' + data.dataInicio + '\n';
                  texto += '📅 *Data Término:* ' + data.dataTermino + '\n';
                  if (data.horaInicio) texto += '⏰ *Início:* ' + data.horaInicio + '\n';
                  if (data.horaTermino) texto += '⏳ *Término:* ' + data.horaTermino + '\n';
                  texto += '📍 *Cidade/Estado:* ' + data.cidadeEstado + '\n';

                  if (data.servicos.length > 0) {
                    texto += '\n*━━━ SERVIÇOS SOLICITADOS ━━━*\n';
                    data.servicos.forEach(function(s) { texto += '✅ ' + s + '\n'; });
                  }

                  if (data.mensagem) {
                    texto += '\n*━━━ OBSERVAÇÕES ━━━*\n' + data.mensagem + '\n';
                  }

                  texto += '\n━━━━━━━━━━━━━━━━━━━━\n';
                  texto += '_Enviado pelo formulário de contato do site._';

                  return 'https://wa.me/5585988601400?text=' + encodeURIComponent(texto);
                }

                // --- Form submit ---
                var form = document.getElementById('ca-form-contato');
                if (form) {
                  form.addEventListener('submit', function(e) {
                    e.preventDefault();

                    var data = {
                      nomeComercial: (document.getElementById('ca-nome-comercial')?.value || '').trim(),
                      responsavel:   (document.getElementById('ca-responsavel')?.value || '').trim(),
                      cargo:         (document.getElementById('ca-cargo')?.value || '').trim(),
                      telefone:      (document.getElementById('ca-telefone')?.value || '').trim(),
                      email:         (document.getElementById('ca-email')?.value || '').trim(),
                      dataInicio:    (document.getElementById('ca-data-inicio')?.value || '').trim(),
                      dataTermino:   (document.getElementById('ca-data-termino')?.value || '').trim(),
                      horaInicio:    (document.getElementById('ca-hora-inicio')?.value || '').trim(),
                      horaTermino:   (document.getElementById('ca-hora-termino')?.value || '').trim(),
                      cidadeEstado:  (document.getElementById('ca-cidade-estado')?.value || '').trim(),
                      mensagem:      (document.getElementById('ca-mensagem')?.value || '').trim(),
                      servicos:      getServicos()
                    };

                    // Validate required fields
                    if (!data.nomeComercial || !data.responsavel || !data.telefone || !data.dataInicio || !data.dataTermino || !data.cidadeEstado) {
                      showToast('⚠️ Preencha todos os campos obrigatórios (*) antes de enviar.', 'error');
                      return;
                    }

                    var btn = document.getElementById('ca-btn-submit');
                    setButtonState(btn, 'loading');
                    showToast('📨 Enviando sua solicitação...');

                    // Fire BOTH in parallel: email + WhatsApp
                    Promise.all([
                      sendEmail(data),
                      Promise.resolve().then(function() {
                        // WhatsApp opens after a short delay so email has time to fire
                        var waUrl = buildWhatsAppUrl(data);
                        setTimeout(function() { window.open(waUrl, '_blank'); }, 800);
                        return { ok: true, channel: 'whatsapp' };
                      })
                    ]).then(function(results) {
                      var emailResult = results[0];

                      if (emailResult && emailResult.success) {
                        showToast('✅ Solicitação enviada por e-mail e WhatsApp! Retornaremos em breve.', 'success');
                      } else if (emailResult && emailResult.skipped) {
                        showToast('✅ Solicitação enviada via WhatsApp! Configure o Web3Forms para ativar o envio por e-mail.', 'success');
                      } else {
                        showToast('⚠️ Erro ao enviar e-mail, mas o WhatsApp foi aberto. Se preferir, envie diretamente para marcos.criaacao@gmail.com', 'error');
                      }

                      setButtonState(btn, 'default');
                    }).catch(function() {
                      showToast('⚠️ Algo deu errado. Tente novamente ou entre em contato diretamente pelo WhatsApp.', 'error');
                      setButtonState(btn, 'default');
                    });
                  });
                }
              })();
            </script>
          </div>

        </div>
      </section>

      <script>
        document.addEventListener("DOMContentLoaded", function () {
          const reveals = document.querySelectorAll(".sol-reveal");
          const revealOnScroll = () => {
            const windowHeight = window.innerHeight;
            const elementVisible = 100;
            reveals.forEach((reveal) => {
              const elementTop = reveal.getBoundingClientRect().top;
              if (elementTop < windowHeight - elementVisible) {
                reveal.classList.add("sol-visible");
              }
            });
          };
          window.addEventListener("scroll", revealOnScroll);
          revealOnScroll();
        });
